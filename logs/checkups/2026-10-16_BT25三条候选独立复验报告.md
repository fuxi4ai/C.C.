# 调度语义（自实现冻结规格）
def schedule(rows):
    out=[]; seen=set(); slots={}
    for x in sorted(rows, key=lambda z:(z['base_i'] if z['base_i'] is not None else 999999, z['key'])):
        if x['base_i'] is None: out.append(x); continue
        exposure = tuple(x['codes']) if x['codes'] else ('unmapped:'+x['theme'],)
        token = (x['base_i'], exposure, x['logic'], x['gap_source'], x['gap_band'])
        if token in seen: continue
        seen.add(token)
        if x['base_i'] < slots.get(x['theme'],-1): continue
        slots[x['theme']] = x['base_i'] + max_delay(x['entry_rule']) + x['max_hold'] + 1
        out.append(x)
    return out
# 收益：成员 pct 逐日复合；net = mean_member − bench − 拖拽系数×(1+asset)
